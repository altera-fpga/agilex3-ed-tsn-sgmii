# ITF describes a high level specification of a regtest using a
# tree data structure. 
#
# Each node in the tree represents a modular test defined by the ITF
# framework or a specific team. Starting at the "itf" root node, we traverse
# and execute the children test nodes until all children are executed.
#
#
# Example:
#                                              rtl_sim_msim_vlg
#                                            /
# itf ---- device ---- qsys_script ---- qsys_generate
#                                            \ 
#                                              qsyn ---- qfit
#
#
# Tests can be highly configured via hooks and options in itf_regtest_local.pl
# Limited configuration can be done in flow.txt
#
# The order of lines in this file is not important.

#The device that is being tested (Agilex 3 / SM)
device=SM7_PART1

# Quartus project name for a5e065bb32aes1_mdk_1x1G VDS (top.qpf / top.qsf)
project=top

# These assignments describe the order the subtest will be executed.
# The "qsys" stage name is historical: it now runs VDS generate
# (scripts/itf/prepare_vds_dut.sh), not Platform Designer qsys-generate.
add_subtest=itf,device
add_subtest=device,qsys
add_subtest=qsys,rtl_sim_compile_only_vcsmx_vlg
add_subtest=rtl_sim_compile_only_vcsmx_vlg,sequence
add_subtest=sequence,rtl_sim_simulate_only_vcsmx

add_subtest=qsys,qsyn
add_subtest=qsyn,qfit
add_subtest=qfit,qsta
# add_subtest=qsta,qsta_timing_info
add_subtest=qsta,qasm
add_subtest=qasm,ocp

# VDS validate/generate + SPD export (prepare_vds_dut.sh) needs more than the 30m default
subtest_timeout=2h,qsys
subtest_timeout=5h,rtl_sim_compile_only_vcsmx_vlg
subtest_timeout=18h,rtl_sim_simulate_only_vcsmx

subtest_memory_requirement=qsys,48000
